#include<stdio.h>
#include<stdlib.h>
#include<string.h>
struct node{
    int data;
    struct node *next;
};
struct node *top=NULL;
    struct node *temp=NULL;
    int count=0;
void push(int x){
    struct node *newnode=(struct node*)malloc(sizeof(struct node));
    
    if(newnode==NULL){
        printf("Stack Overflow\n");
    }
    newnode->data=x;
    newnode->next=top;
    top=newnode;
}
void pop(){
    if(top==NULL){
        printf("Stack Underflow\n");
    }
    
    temp=top;
    top=top->next;
    free(temp);
}
void size(){
    temp=top;
    while(temp!=NULL){
        count++;
        temp=temp->next;
    }
    printf("%d\n",count);
}
void display(){
    if(top==NULL){
                printf("Stack Underflow\n");
    }
    temp=top;
    while(temp!=NULL){
        printf("%d\n",temp->data);
        temp=temp->next;
    }
}
void peek(){
   printf("%d\n",top->data); 
}
void isempty(){
    if(top==NULL){
        printf("True\n");
    }
    else {
        printf("False\n");
    }
}
int main(){
    int n,x;
    scanf("%d",&n);
    char op[12];
    for(int i=0;i<n;i++){
        scanf("%s",op);
        if(strcmp(op,"push")==0){
            scanf("%d",&x);
            push(x);
        }
       else if(strcmp(op,"pop")==0){
            
            pop();
        }
        else if(strcmp(op,"display")==0){
            
            display();
        }
        else if(strcmp(op,"peek")==0){
            
            peek();
        }
        else if(strcmp(op,"size")==0){
            
            size();
        }
        else if(strcmp(op,"isempty")==0){
            
            isempty();
        }
    }
}
