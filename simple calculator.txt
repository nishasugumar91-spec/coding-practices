#include<stdio.h>
#include<stdlib.h>
#include<string.h>
#include<ctype.h>
#define MAX 100
int s[MAX];
int top=-1;
void push(int  x){
    s[++top]=x;
}
int pop(){
     return s[top--];
}
int main(){
    char input[MAX];
    int result;
  fgets(input, sizeof(input), stdin);
    char *token=strtok(input," \n ");
    while(token!=NULL){
        if(isdigit(token[0])){
            int num = atoi(token);
            push(num);
        }
        else {
            int b=pop();
            int a=pop();
            if(strcmp(token,"+")==0){
                 result=a+b;
            }
            else if(strcmp(token,"-")==0){
                 result=a-b;
            }
            else if(strcmp(token,"*")==0){
                 result=a*b;
            }
            else if(strcmp(token,"/")==0){
                 result=a/b;
            }
        push(result);}
        token=strtok(NULL," \n ");
    }
    printf("%d",pop());
}
